function esc(t){return String(t||'').replace(/&/g,'&').replace(//g,'>').replace(/"/g,'"');}

// Safe markdown: never injects raw HTML from responses
function safeMarkdown(text){
  // Escape ALL html first, then selectively add safe tags
  var t=esc(text);
  t=t.replace(/^### (.+)$/gm,'
$1
');
  t=t.replace(/^## (.+)$/gm,'
$1
');
  t=t.replace(/^# (.+)$/gm,'
$1
');
  t=t.replace(/\*\*(.+?)\*\*/g,'$1');
  t=t.replace(/\*(.+?)\*/g,'$1');
  t=t.replace(/`([^`]+)`/g,'$1');
  t=t.replace(/^[-*] (.+)$/gm,'
  • $1
    • ');
  t=t.replace(/^(\d+)\. (.+)$/gm,'
  • $2
    • ');
  t=t.replace(/\n\n/g,'
    ').replace(/\n/g,'
    ');
  return '
    '+t+'
    ';
}

// Parse response: extract code blocks, return {before, code, lang, after}
function parseResponse(text){
  var m=text.match(/```(\w+)?\r?\n?([\s\S]*?)```/);
  if(m){
    var before=text.slice(0,m.index).trim();
    var after=text.slice(m.index+m[0].length).trim();
    return{hasCode:true,before:before,code:m[2].trim(),lang:(m[1]||'html').toLowerCase(),after:after};
  }
  return{hasCode:false,before:text.trim(),code:'',lang:'',after:''};
}

function esc(t){return String(t||'').replace(/&/g,'&').replace(//g,'>').replace(/"/g,'"');}

// Safe markdown: never injects raw HTML from responses
function safeMarkdown(text){
  // Escape ALL html first, then selectively add safe tags
  var t=esc(text);
  t=t.replace(/^### (.+)$/gm,'
    $1
    ');
  t=t.replace(/^## (.+)$/gm,'
    $1
    ');
  t=t.replace(/^# (.+)$/gm,'
    $1
    ');
  t=t.replace(/\*\*(.+?)\*\*/g,'$1');
  t=t.replace(/\*(.+?)\*/g,'$1');
  t=t.replace(/`([^`]+)`/g,'$1');
  t=t.replace(/^[-*] (.+)$/gm,'
  • $1
    • ');
  t=t.replace(/^(\d+)\. (.+)$/gm,'
  • $2
    • ');
  t=t.replace(/\n\n/g,'
    ').replace(/\n/g,'
    ');
  return '
    '+t+'
    ';
}

// Parse response: extract code blocks, return {before, code, lang, after}
function parseResponse(text){
  var m=text.match(/```(\w+)?\r?\n?([\s\S]*?)```/);
  if(m){
    var before=text.slice(0,m.index).trim();
    var after=text.slice(m.index+m[0].length).trim();
    return{hasCode:true,before:before,code:m[2].trim(),lang:(m[1]||'html').toLowerCase(),after:after};
  }
  return{hasCode:false,before:text.trim(),code:'',lang:'',after:''};
}
